probs = np.array([0.30, 0.25, 0.20, 0.15, 0.10])
words = ["A", "B", "C", "D", "E"]

# Top-k
k = 2
top_k_indices = probs.argsort()[-k:]
top_k_probs = probs[top_k_indices]
top_k_probs = top_k_probs / top_k_probs.sum()
