from transformers import GPT2Tokenizer

tokenizer = GPT2Tokenizer.from_pretrained("gpt2")

text = "I love python and transformers!"


tokens = tokenizer.tokenize(text)
token_ids = tokenizer.convert_tokens_to_ids(tokens)

print("original text", text)
print("tokens", tokens)
print("token ids:", token_ids)

decoded = tokenizer.decode(token_ids)
print(decoded)

documents = [
    "Python is a programming language used for AI and web development.",
    "RAG means Retrieval-Augmented Generation.",
    "In RAG, the system first searches documents and then generates an answer.",
    "Vector RAG databases store text in numerical form for semantic search."

# Simple keyword search
relevant_docs = [RAG means Retrieval-Augmented Generation,In RAG, the system first searches documents and then generates an answer.,"Vector RAG databases store text in numerical form for semantic search."]

        relevant_docs.append(doc)
